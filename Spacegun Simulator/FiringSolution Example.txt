using System;

public struct Vector2
{
    public float X;
    public float Y;

    public Vector2(float x, float y)
    {
        X = x;
        Y = y;
    }
}

public class FiringSolutionResult
{
    public bool CanDestroy { get; set; }
    public bool CanHit { get; set; }
    public bool SolutionValid { get; set; }
    public Vector2? InterceptPoint { get; set; }
    public float TimeToImpact { get; set; }
    public float FiringAngle { get; set; }
    public float MinVelocityRequired { get; set; }
    public float ProjectileVelocity { get; set; }
    public string Message { get; set; }
}

public class FiringSolution
{
    private float projectileMass;
    private float enemyFractureEnergy;

    public FiringSolution(float projectileMass, float enemyFractureEnergy)
    {
        this.projectileMass = projectileMass;
        this.enemyFractureEnergy = enemyFractureEnergy;
    }

    public float CalculateRequiredVelocity()
    {
        // Calculate minimum projectile velocity needed to destroy enemy
        // Using KE = 0.5 * m * v^2, solve for v
        float requiredVelocity = (float)Math.Sqrt(2 * enemyFractureEnergy / projectileMass);
        return requiredVelocity;
    }

    public (Vector2? interceptPoint, float timeToImpact, bool canHit) CalculateIntercept(
        float currentDistance, Vector2 enemyVelocity, float projectileVelocity)
    {
        // Enemy position (assuming enemy starts at current distance)
        float enemyX = currentDistance;
        float enemyY = 0;

        // Enemy velocity components
        float vx = enemyVelocity.X;
        float vy = enemyVelocity.Y;
        float enemySpeed = (float)Math.Sqrt(vx * vx + vy * vy);

        // Check if projectile is fast enough to catch enemy
        if (projectileVelocity <= enemySpeed && vx < 0)
        {
            // Enemy moving away faster than projectile
            return (null, 0, false);
        }

        // Solve quadratic equation for intercept time
        // Enemy position at time t: (enemyX + vx*t, enemyY + vy*t)
        // Projectile must travel distance: projectileVelocity * t
        // Distance equation: sqrt((enemyX + vx*t)^2 + (enemyY + vy*t)^2) = projectileVelocity * t

        // Expanding: (enemyX + vx*t)^2 + (enemyY + vy*t)^2 = (projectileVelocity * t)^2
        // This becomes: a*t^2 + b*t + c = 0

        float a = vx * vx + vy * vy - projectileVelocity * projectileVelocity;
        float b = 2 * (enemyX * vx + enemyY * vy);
        float c = enemyX * enemyX + enemyY * enemyY;

        // Solve quadratic
        float discriminant = b * b - 4 * a * c;

        if (discriminant < 0)
        {
            return (null, 0, false); // No solution
        }

        float t;
        if (Math.Abs(a) < 1e-10f) // Nearly zero, linear equation
        {
            if (Math.Abs(b) < 1e-10f)
            {
                return (null, 0, false);
            }
            t = -c / b;
        }
        else
        {
            float t1 = (-b + (float)Math.Sqrt(discriminant)) / (2 * a);
            float t2 = (-b - (float)Math.Sqrt(discriminant)) / (2 * a);

            // Take earliest positive time
            if (t1 > 0 && t2 > 0)
            {
                t = Math.Min(t1, t2);
            }
            else if (t1 > 0)
            {
                t = t1;
            }
            else if (t2 > 0)
            {
                t = t2;
            }
            else
            {
                return (null, 0, false);
            }
        }

        // Calculate intercept point
        float interceptX = enemyX + vx * t;
        float interceptY = enemyY + vy * t;

        return (new Vector2(interceptX, interceptY), t, true);
    }

    public float GetFiringAngle(Vector2 interceptPoint, Vector2? origin = null)
    {
        // Calculate angle to aim at intercept point
        // Returns angle in degrees (0 = right, 90 = up)
        Vector2 originPoint = origin ?? new Vector2(0, 0);
        float dx = interceptPoint.X - originPoint.X;
        float dy = interceptPoint.Y - originPoint.Y;
        float angleRad = (float)Math.Atan2(dy, dx);
        float angleDeg = angleRad * 180f / (float)Math.PI;
        return angleDeg;
    }

    public FiringSolutionResult FullSolution(
        float currentDistance, Vector2 enemyVelocity, float projectileVelocity)
    {
        // Check if projectile has enough energy
        float minVelocity = CalculateRequiredVelocity();
        bool hasEnergy = projectileVelocity >= minVelocity;

        // Calculate intercept
        var (interceptPoint, timeToImpact, canHit) = CalculateIntercept(
            currentDistance, enemyVelocity, projectileVelocity);

        if (!canHit)
        {
            return new FiringSolutionResult
            {
                CanDestroy = hasEnergy,
                CanHit = false,
                SolutionValid = false,
                MinVelocityRequired = minVelocity,
                ProjectileVelocity = projectileVelocity,
                Message = "Cannot intercept target with current velocity"
            };
        }

        float firingAngle = GetFiringAngle(interceptPoint.Value);

        return new FiringSolutionResult
        {
            CanDestroy = hasEnergy,
            CanHit = canHit,
            SolutionValid = hasEnergy && canHit,
            InterceptPoint = interceptPoint,
            TimeToImpact = timeToImpact,
            FiringAngle = firingAngle,
            MinVelocityRequired = minVelocity,
            ProjectileVelocity = projectileVelocity,
            Message = (hasEnergy && canHit) ? "Valid firing solution" : "Insufficient velocity to destroy target"
        };
    }
}

// Example usage
class Program
{
    static void Main()
    {
        // Game parameters
        float projectileMass = 5.0f; // kg
        float enemyFractureEnergy = 50000f; // joules

        // Create calculator
        FiringSolution calculator = new FiringSolution(projectileMass, enemyFractureEnergy);

        // Scenario 1: Enemy moving diagonally
        Console.WriteLine("=== Scenario 1: Enemy approaching diagonally ===");
        float currentDistance = 1000f; // meters
        Vector2 enemyVelocity = new Vector2(-50f, 30f); // moving toward us and upward (m/s)
        float projectileVelocity = 300f; // m/s

        FiringSolutionResult solution = calculator.FullSolution(currentDistance, enemyVelocity, projectileVelocity);

        Console.WriteLine($"Can destroy: {solution.CanDestroy}");
        Console.WriteLine($"Can hit: {solution.CanHit}");
        Console.WriteLine($"Solution valid: {solution.SolutionValid}");
        if (solution.CanHit && solution.InterceptPoint.HasValue)
        {
            Console.WriteLine($"Intercept point: ({solution.InterceptPoint.Value.X:F2}, {solution.InterceptPoint.Value.Y:F2})");
            Console.WriteLine($"Time to impact: {solution.TimeToImpact:F2} seconds");
            Console.WriteLine($"Firing angle: {solution.FiringAngle:F2} degrees");
        }
        Console.WriteLine($"Min velocity required: {solution.MinVelocityRequired:F2} m/s");
        Console.WriteLine($"Message: {solution.Message}\n");

        // Scenario 2: Enemy moving away too fast
        Console.WriteLine("=== Scenario 2: Enemy escaping ===");
        currentDistance = 1000f;
        enemyVelocity = new Vector2(350f, 0f); // moving away faster than projectile
        projectileVelocity = 300f;

        solution = calculator.FullSolution(currentDistance, enemyVelocity, projectileVelocity);
        Console.WriteLine($"Solution valid: {solution.SolutionValid}");
        Console.WriteLine($"Message: {solution.Message}\n");

        // Scenario 3: Insufficient projectile velocity
        Console.WriteLine("=== Scenario 3: Insufficient energy ===");
        currentDistance = 500f;
        enemyVelocity = new Vector2(20f, -10f);
        projectileVelocity = 100f; // Too slow to destroy

        solution = calculator.FullSolution(currentDistance, enemyVelocity, projectileVelocity);
        Console.WriteLine($"Can hit: {solution.CanHit}");
        Console.WriteLine($"Can destroy: {solution.CanDestroy}");
        Console.WriteLine($"Solution valid: {solution.SolutionValid}");
        Console.WriteLine($"Message: {solution.Message}");
    }
}
